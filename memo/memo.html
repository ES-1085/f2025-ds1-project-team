mlb_to_mmbtu      <- 1.19
therm_to_mmbtu    <- 0.1

gal_no56_to_mmbtu <- 0.150

# MMBtu → kg CO₂
kg_per_mmbtu_gas   <- 53.06
kg_per_mmbtu_oil2  <- 73.16

kg_per_mmbtu_oil56 <- 78.80

Step 4: Classify Fuel Types

carbon_ready <- carbon_ready |>
  mutate(
    fuel = case_when(
      str_detect(measurement, "(?i)electric") ~ "electricity",
      str_detect(measurement, "(?i)steam")    ~ "steam",
      str_detect(measurement, "(?i)gas|therm")~ "natural_gas",
      str_detect(measurement, "(?i)oil.*2")   ~ "oil_2",
      str_detect(measurement, "(?i)oil.*4")   ~ "oil_4",
      str_detect(measurement, "(?i)oil.*(5|6)") ~ "oil_56",
      str_detect(measurement, "(?i)total usage") ~ "total_mmbtu",
      TRUE ~ "other"

      levels = c("1","2","3","4","5"),

carbon_by_row <- carbon_ready |>
  mutate(
    kg_co2 = case_when(
      fuel == "electricity" ~ total_raw * kg_per_kwh_grid,

      fuel == "steam"       ~ total_raw *
                                mlb_to_mmbtu *
                                kg_per_mmbtu_gas,

      fuel == "natural_gas" ~ total_raw *
                                therm_to_mmbtu *
                                kg_per_mmbtu_gas,

      fuel == "oil_2"       ~ total_raw *
                                gal_no2_to_mmbtu *
                                kg_per_mmbtu_oil2,

      fuel == "oil_4"       ~ total_raw *
                                gal_no4_to_mmbtu *
                                kg_per_mmbtu_oil4,

      fuel == "oil_56"      ~ total_raw *
                                gal_no56_to_mmbtu *
                                kg_per_mmbtu_oil56,

      TRUE ~ NA_real_
    )
  )
